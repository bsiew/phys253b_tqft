% -------- Theorem environments --------
%
% Brian's \defthm pattern from references/style references/hdr.tex
% and springer.tex. We share the equation counter, so equations and
% theorems interleave within a section.
%
% Environments listed below are the ones Brian actually reaches for.
% The four canonical AI-tell environments (upshot, note, slogan,
% motivation) and four adjacent ones (warning, idea, intuition,
% terminology) are DELIBERATELY omitted. See plan/README.md rule #2.

\theoremstyle{definition}
\newtheorem{theorem}[equation]{Theorem}
\numberwithin{equation}{section}

\usepackage{mfirstuc}
\newcommand{\capitalizename}[1]{\makefirstuc{#1}}
\newcommand{\defthm}[1]{\newtheorem{#1}[equation]{\capitalizename{#1}}}
\newcommand{\defthms}[1]{\forcsvlist{\defthm}{#1}}

\defthms{%
  assumption,claim,construction,corollary,definition,example,
  exercise,fact,lemma,
  notation,proposition,question,remark,setup%
}

% -------- Unique hyperref anchors under report-class chaptering --------
%
% The CS standalone source uses \thesection.\arabic{equation} (article
% class). In report-class, equation 5.1.1 of Chapter 5 and equation
% 5.1.1 of Chapter 10 would collide as hyperref anchors if we didn't
% prepend \thechapter. This block fixes that.

\makeatletter
\renewcommand*\theHequation{\thechapter.\thesection.\arabic{equation}}
\let\theHtheorem\theHequation
\let\theHassumption\theHequation
\let\theHclaim\theHequation
\let\theHconstruction\theHequation
\let\theHcorollary\theHequation
\let\theHdefinition\theHequation
\let\theHexample\theHequation
\let\theHexercise\theHequation
\let\theHfact\theHequation
\let\theHlemma\theHequation
\let\theHnotation\theHequation
\let\theHproposition\theHequation
\let\theHquestion\theHequation
\let\theHremark\theHequation
\let\theHsetup\theHequation
\makeatother
